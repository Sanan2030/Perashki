export default async function handler(req, res) {
  try {
    const method = req.method || 'GET';
    const action = route[method];
    if (typeof action !== 'function') {
      res.writeHead(405, { Allow: Object.keys(route).filter(key => /^(GET|POST|PUT|DELETE|PATCH|HEAD|OPTIONS)$/.test(key)).join(', ') });
      res.end(); return;
    }
    const chunks = [];
    let size = 0;
    for await (const chunk of req) {
      size += chunk.length;
      if (size > 1_000_000) { res.writeHead(413); res.end(); return; }
      chunks.push(chunk);
    }
    const headers = new Headers();
    for (const [name, value] of Object.entries(req.headers)) {
      if (value !== undefined) headers.set(name, Array.isArray(value) ? value.join(', ') : value);
    }
    const protocol = req.headers['x-forwarded-proto'] || 'https';
    const request = new Request(new URL(req.url, `${protocol}://${req.headers.host || 'localhost'}`), {
      method, headers,
      ...(!['GET', 'HEAD'].includes(method) ? { body: Buffer.concat(chunks) } : {}),
    });
    const response = await action(request);
    res.writeHead(response.status, Object.fromEntries(response.headers));
    res.end(Buffer.from(await response.arrayBuffer()));
  } catch (error) {
    console.error('Application API failed:', error);
    if (!res.headersSent) res.writeHead(500, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: 'Sorğu tamamlanmadı.' }));
  }
}
